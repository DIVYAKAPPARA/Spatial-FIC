compute_FIC <-function(y,X,w1,p,beta,ns=ns,c_lev){

  
  coef=slm$coefficients
  
  #summary(slm)
  
  Vcov_O=round(slm$resvar,4)
  
  ###change the order rho, beta and sigma^2 
  
  Vcov_O=Vcov_O[c(2:nrow(Vcov_O),1),c(2:ncol(Vcov_O),1)]
  
  ##Hessian matrix
  
  Hess_mat=solve(Vcov_O)
  
  
  ##partitions
  
  I_rr= as.matrix(Hess_mat[1,1])
  I_rb=as.matrix(Hess_mat[1,c(2:(p+1))]); I_br=t(I_rb)
  I_bb= as.matrix(Hess_mat[c(2:(p+1)),c(2:(p+1))])
  I_rs= as.matrix(Hess_mat[1,(p+2)]); 
  I_bs=as.matrix(Hess_mat[c(2:(p+1)),(p+2)]); I_sb=t(I_bs)
  I_ss=as.matrix(Hess_mat[(p+2),(p+2)])
  
  #int=rep(1,nrow(w1))
  #X_in=cbind(int,X)
  
  
  I_S_ls=I_Sinv_ls=list()
  FIC=Bias=Variance=Mse1=NULL; sm=list()
  for(i in 1:length(ns)){

    
    
    I_S12=t(I_rb)%*%t(proj_ls[[i]])
    I_S21=(proj_ls[[i]])%*%t(I_br)
    I_S22=proj_ls[[i]]%*%I_bb%*%t(proj_ls[[i]])
    I_S32=t(c(rep(0,ns[i])))
    I_S23=t(I_S32)
    
    #I_S=matrix(c(I_rr,I_S12,I_rs,I_S21,I_S22,I_S23,I_rs,I_S32,I_ss),nrow=5,ncol=5)
    I_S=matrix(0,nrow=ns[i]+2,ncol=ns[i]+2)
    I_S[1,1]=I_rr; I_S[1,2:(ns[i]+1)]=I_S12; I_S[1,nrow(I_S)]=I_rs
    I_S[2:(ns[i]+1),1]=I_S21; I_S[2:(ns[i]+1),2:(ns[i]+1)]=I_S22; I_S[2:(ns[i]+1),nrow(I_S)]=I_S23
    I_S[nrow(I_S),1]=I_rs; I_S[nrow(I_S),2:(ns[i]+1)]=I_S32; I_S[nrow(I_S),nrow(I_S)]=I_ss
    
    I_S_ls[[i]]=I_S
    I_Sinv_ls[[i]]=solve(I_S)
    
    m1=I_Sinv_ls[[i]][1,1]%*%t(I_rb)+I_Sinv_ls[[i]][1,2:(ns[i]+1)]%*%proj_ls[[i]]%*%I_bb
    m2=(I_Sinv_ls[[i]][2:(ns[i]+1),1])%*%t(I_rb)+I_Sinv_ls[[i]][2:(ns[i]+1),2:(ns[i]+1)]%*%proj_ls[[i]]%*%I_bb
    m3=I_Sinv_ls[[i]][(ns[i]+2),1]%*%t(I_rb)
    
    J=c(w1[c_lev,]%*%y,(X%*%t(proj_ls[[i]]))[c_lev,],0)
    D_n=sqrt(nrow(w1))%*%slm$coefficients ; delta_dt=t(D_n)%*%(D_n)
    B_ws=J[1]%*%m1+J[2:(ns[i]+1)]%*%m2+J[(ns[i]+2)]%*%m3
    V_ws=t(J)%*%I_Sinv_ls[[i]]%*%J
    
    Bias[i]=(B_ws)%*%delta_dt%*%t(B_ws); Variance[i]=V_ws
    FIC[i]=(B_ws)%*%delta_dt%*%t(B_ws)+V_ws
    Mse1[i]=((((X)%*%t(proj_ls[[i]]))[c_lev,]%*%(proj_ls[[i]]%*%coef)-y[c_lev])^2)
    
    
  }
  return(list(FIC=FIC,Mse1=Mse1,X=X,smodel=sm,bias=Bias,Vr=Variance,Beta_est=coef,rho_hat=slm$rho))
  
} #linear predictor

AFIC=function(w1,p,beta,rho,ns,psi){
  mu=rep(0,p); Sigma=diag(1,p)
  X=mvrnorm(n = nrow(w1), mu,Sigma=Sigma , tol = 1e-6, empirical = FALSE, EISPACK = FALSE)
  rm(mu,Sigma)
  #int=rep(1,nrow(w1))
  #X=cbind(rep(1,nrow(w1)),X)
  #colnames(X)=NULL; X=as.matrix(X)
  
  y_hat=solve(diag(nrow(w1)) - rho * w1) %*% (crossprod(t(X), beta)+rnorm(nrow(w1),0,1))
  slm <- lagsarlm(y_hat ~X,listw = W1,method = "eigen") ##
  coef=slm$coefficients
  #summary(slm)
  
  Vcov_O=round(slm$resvar,4)
  
  ###change the order rho, beta and sigma^2 
  
  Vcov_O=Vcov_O[c(2:nrow(Vcov_O),1),c(2:ncol(Vcov_O),1)]
  
  ##Hessian matrix
  
  Hess_mat=solve(Vcov_O)
  
  ##partitions
  
  I_rr= as.matrix(Hess_mat[1,1])
  I_rb=as.matrix(Hess_mat[1,c(2:(p+2))]); I_br=t(I_rb)
  I_bb= as.matrix(Hess_mat[c(2:(p+2)),c(2:(p+2))])
  I_rs= as.matrix(Hess_mat[1,(p+3)]); 
  I_bs=as.matrix(Hess_mat[c(2:(p+2)),(p+3)]); I_sb=t(I_bs)
  I_ss=as.matrix(Hess_mat[(p+3),(p+3)])
  int=rep(1,nrow(w1))
  X_in=cbind(int,X)
  
  H_nw=matrix(0,p+2,p+2)
  H_nw[1,1]=t(w1%*%y_hat)%*%psi%*%(w1%*%y_hat); H_nw[1,c(2:(p+2))]=t(w1%*%y_hat)%*%psi%*%X_in
  H_nw[c(2:(p+2)),1]=t(X_in)%*%psi%*%(w1%*%y_hat); H_nw[c(2:(p+2)),c(2:(p+2))]=t(X_in)%*%psi%*%X_in
  
  K_w=I_rb%*%(1/I_rr)%*%H_nw[1,1]%*%(1/I_rr)%*%(I_br)-2*I_rb%*%(1/I_rr)%*%H_nw[1,c(2:(p+2))]+H_nw[c(2:(p+2)),c(2:(p+2))]
  Q=solve(I_bb-I_rb%*%(1/I_rr)%*%I_br)
    
  I_S_ls=I_Sinv_ls=list()
  AFIC=NULL; sm=list()
  for(i in 1:length(ns)){
    
    I_S12=t(I_rb)%*%t(proj_ls[[i]])
    I_S21=(proj_ls[[i]])%*%t(I_br)
    I_S22=proj_ls[[i]]%*%I_bb%*%t(proj_ls[[i]])
#    I_S32=t(c(rep(0,ns[i])))
#    I_S23=t(I_S32)
    
    #I_S=matrix(c(I_rr,I_S12,I_rs,I_S21,I_S22,I_S23,I_rs,I_S32,I_ss),nrow=5,ncol=5)
    I_S=matrix(0,nrow=ns[i]+1,ncol=ns[i]+1)
    I_S[1,1]=I_rr; I_S[1,2:(ns[i]+1)]=I_S12
    I_S[2:(ns[i]+1),1]=I_S21; I_S[2:(ns[i]+1),2:(ns[i]+1)]=I_S22
#    I_S[nrow(I_S),1]=I_rs; I_S[nrow(I_S),2:(ns[i]+1)]=I_S32; I_S[nrow(I_S),nrow(I_S)]=I_ss
    
    I_S_ls[[i]]=I_S
    I_Sinv_ls[[i]]=solve(I_S)
    
    Q_s=solve(proj_ls[[i]]%*%solve(Q)%*%t(proj_ls[[i]]))
    G_s=t(proj_ls[[i]])%*%Q_s%*%proj_ls[[i]]%*%solve(Q)
    D_n=sqrt(nrow(w1))%*%slm$coefficients ; delta_dt=t(D_n)%*%(D_n)
    AFIC[i]=tr((diag(1,p+1)-G_s)%*%delta_dt%*%t(diag(1,p+1)-G_s)%*%K_w)+tr(G_s%*%Q%*%t(G_s)%*%K_w)
  }
  
  # true data generating process
  return(list(AFIC=AFIC,x=X_in,yhat=y_hat,smodel=sm,Beta_est=coef,rho_hat=slm$rho))
}

# Function to compute Gaussian Kernel Weight Matrix
gaussian_kernel_weights <- function(X, fixed_index, h) {
  # Number of regions (n) and covariates (p)
  n <- nrow(X)
  diff_X=psi_values=NULL
  # Extract the fixed covariate row
  X_fixed <- X[fixed_index, , drop = FALSE]
  
  # Compute Euclidean distances ||X_i - X_fixed||
  for(i in 1:n){
    diff_X[i] <- sqrt(sum(X_fixed - X[i,])^2) / h
    
    # Gaussian Kernel function
    psi_values[i] <- (1 / sqrt(2 * pi)) * exp(-0.5 * diff_X[i]^2)
  }
  # Construct diagonal weight matrix
  Psi <- diag(psi_values, n, n)
  
  return(Psi)
}
